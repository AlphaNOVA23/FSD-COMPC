-- 2. Employee Table (Base Entity)
CREATE TABLE employee (
    employee_id SERIAL PRIMARY KEY,
    employee_name VARCHAR(100)
);

-- 3. Department Head Table (Must be created before Department to avoid circular dependency issues during insertion, 
-- or created after and linked via ALTER. Here we create it first with nullable department_id initially if needed, 
-- but strictly speaking, circular FKs usually require creating tables first then adding constraints. 
-- For simplicity based on your previous script, I will create tables then add FKs where circular.)

CREATE TABLE department_head (
    head_id SERIAL PRIMARY KEY,
    employee_id INTEGER UNIQUE REFERENCES employee(employee_id),
    head_role VARCHAR(100),
    head_term VARCHAR(50)
    -- department_id will be added/linked below
);

-- 4. Department Table
CREATE TABLE department (
    department_id SERIAL PRIMARY KEY,
    department_name VARCHAR(100) NOT NULL,
    department_location VARCHAR(255),
    head_id INTEGER, -- FK to department_head
    department_capacity INTEGER,
    department_contact VARCHAR(255)
);

-- 5. Add Constraints for Department <-> Department Head
-- Link Department to its Head
ALTER TABLE department
    ADD CONSTRAINT fk_department_head
    FOREIGN KEY (head_id) REFERENCES department_head(head_id);

-- Link Head to their Department (Circular)
ALTER TABLE department_head
    ADD COLUMN department_id INTEGER UNIQUE;

ALTER TABLE department_head
    ADD CONSTRAINT fk_head_department
    FOREIGN KEY (department_id) REFERENCES department(department_id);

-- 6. Salaries Table (The New Integrated Module)
CREATE TABLE salaries (
    salary_id SERIAL PRIMARY KEY,
    employee_id INTEGER NOT NULL UNIQUE, -- One salary record per employee
    base_salary DOUBLE PRECISION,
    bonus DOUBLE PRECISION,
    deductions DOUBLE PRECISION,
    net_salary DOUBLE PRECISION,
    
    CONSTRAINT fk_salary_employee 
        FOREIGN KEY (employee_id) 
        REFERENCES employee(employee_id)
        ON DELETE CASCADE
);

-- 8. Shift Table
CREATE TABLE shift (
    shift_id SERIAL PRIMARY KEY,
    shift_type VARCHAR(50) NOT NULL, -- Morning, Evening, etc.
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    break_duration INTEGER -- in minutes
);

-- 9. Timesheet Table
CREATE TABLE timesheet (
    timesheet_id SERIAL PRIMARY KEY,
    employee_id INTEGER NOT NULL,
    shift_id INTEGER NOT NULL,
    work_date DATE NOT NULL,
    check_in TIMESTAMP,
    check_out TIMESTAMP,
    total_hours DECIMAL(5,2),
    overtime_hours DECIMAL(5,2),
    status VARCHAR(50) NOT NULL, -- Approved, Pending, etc.
    remarks TEXT,
    
    -- Foreign Keys
    CONSTRAINT fk_timesheet_employee 
        FOREIGN KEY (employee_id) 
        REFERENCES employee(employee_id)
        ON DELETE CASCADE,
        
    CONSTRAINT fk_timesheet_shift 
        FOREIGN KEY (shift_id) 
        REFERENCES shift(shift_id)
);

-- 11. Position Details Table (1-to-1 with Employee)

    job_level VARCHAR(50) NOT NULL, -- Enum stored as string

    CONSTRAINT fk_position_employee
        FOREIGN KEY (employee_id)
        REFERENCES employee(employee_id)
        ON DELETE CASCADE
);

-- 21. Performance Feedback Table
CREATE TABLE performance_feedback (
    feedback_id SERIAL PRIMARY KEY,
    
    -- Who is being reviewed
    employee_id INTEGER NOT NULL,
    
    -- Who is giving the review
    reviewer_id INTEGER NOT NULL,
    
    rating NUMERIC(3, 1) NOT NULL, -- Allows 4.5, etc.
    comments TEXT,
    feedback_date DATE NOT NULL,
    
    -- Loose coupling for modules not yet standardized
    evaluation_id INTEGER,
    training_id INTEGER,

    CONSTRAINT fk_feedback_employee
        FOREIGN KEY (employee_id)
        REFERENCES employee(employee_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_feedback_reviewer
        FOREIGN KEY (reviewer_id)
        REFERENCES employee(employee_id)
        ON DELETE CASCADE
);

-- 23. Performance Evaluation Table (The Master Record)
-- This table matches the standardized Java Entity 'PerformanceEvaluationEntity'
CREATE TABLE performance_evaluation (
    evaluation_id SERIAL PRIMARY KEY,
    employee_id INTEGER NOT NULL,
    evaluation_period VARCHAR(100) NOT NULL,
    performance_rating VARCHAR(50),
    comments TEXT,
    
    CONSTRAINT fk_evaluation_employee 
        FOREIGN KEY (employee_id) 
        REFERENCES employee(employee_id)
        ON DELETE CASCADE
);

-- 24. Link Feedback to Evaluation (Foreign Key Update)
-- Your existing table 'performance_feedback' (21) had a loose column 'evaluation_id'.
-- Now that we created the table, let's enforce the relationship.

ALTER TABLE performance_feedback
    ADD CONSTRAINT fk_feedback_evaluation
    FOREIGN KEY (evaluation_id)
    REFERENCES performance_evaluation(evaluation_id)
    ON DELETE SET NULL;

-- 26. Final Verification Query
SELECT 
    e.employee_name,
    pe.evaluation_period,
    pe.performance_rating,
    COUNT(pf.feedback_id) as feedback_count
FROM performance_evaluation pe
JOIN employee e ON pe.employee_id = e.employee_id
LEFT JOIN performance_feedback pf ON pe.evaluation_id = pf.evaluation_id
GROUP BY e.employee_name, pe.evaluation_period, pe.performance_rating;
